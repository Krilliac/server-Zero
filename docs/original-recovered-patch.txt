+void AnticheatMgr::HandleCheat(Player* player, CheatType cheatType)
+{
+    LogCheat(player, cheatType);
+
+    if (sWorld->getConfig(CONFIG_BOOL_DEBUG_VISUALIZERS))
+    {
+        sDebugVisualizer->LogCheatEvent(player, cheatType);
+    }
+
+    switch (cheatType)
+    {
+        case CHEAT_TYPE_POSITION_HACK:
+        case CHEAT_TYPE_FALL_HACK:
+        {
+            player->TeleportTo(
+                player->m_homebindMapId,
+                player->m_homebindX,
+                player->m_homebindY,
+                player->m_homebindZ,
+                0
+            );
+            break;
+        }
+
+        case CHEAT_TYPE_SPEED_HACK:
+        {
+            player->GetSession()->KickPlayer();
+            break;
+        }
+
+        case CHEAT_TYPE_TIME_DESYNC:
+        {
+            ApplyCounterMeasure(player, COUNTER_MEASURE_TIME_RESET);
+            break;
+        }
+
+        case CHEAT_TYPE_LATENCY_MANIPULATION:
+        {
+            ApplyCounterMeasure(player, COUNTER_MEASURE_LATENCY_RESET);
+            break;
+        }
+
+        case CHEAT_TYPE_TERRAIN_HACK:
+        {
+            ApplyCounterMeasure(player, COUNTER_MEASURE_TELEPORT);
+            break;
+        }
+
+        default:
+            break;
+    }
+}
+
+void AnticheatMgr::ApplyCounterMeasure(Player* player, CounterMeasureType type)
+{
+    if (!player)
+        return;
+
+    switch (type)
+    {
+        case COUNTER_MEASURE_TELEPORT:
+        {
+            player->TeleportTo(
+                player->m_homebindMapId,
+                player->m_homebindX,
+                player->m_homebindY,
+                player->m_homebindZ,
+                0
+            );
+            break;
+        }
+
+        case COUNTER_MEASURE_TIME_RESET:
+        {
+            WorldPacket data(SMSG_FORCE_TIME_UPDATE, 4);
+            data << uint32(time(nullptr));
+            player->GetSession()->SendPacket(&data);
+            break;
+        }
+
+        case COUNTER_MEASURE_LATENCY_RESET:
+        {
+            player->GetSession()->ResetLatency();
+            break;
+        }
+
+        case COUNTER_MEASURE_TEMP_BAN:
+        {
+            break;
+        }
+
+        default:
+            break;
+    }
+}
